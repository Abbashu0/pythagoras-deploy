   221→function summarizeSelectedOptions(selectedIds, options) {
   222→  const available = Array.isArray(options) ? options : [];
   223→  const selected = available.filter((option) => selectedIds.includes(option.id));
   224→
   225→  if (!available.length || selected.length === available.length) {
   226→    return "الكل";
   227→  }
   228→
   229→  if (selected.length === 1) {
   230→    return selected[0].label;
   231→  }
   232→
   233→  if (selected.length === 2) {
   234→    return `${selected[0].label} + ${selected[1].label}`;
   235→  }
   236→
   237→  return `${selected.length} محددة`;
   238→}

   367→function bindQuestionBankStickyState() {

   369→
   370→  const sticky = document.querySelector("[data-question-search-sticky]");
   371→  const sentinel = document.querySelector("[data-question-search-sentinel]");

   378→  questionBankStickyObserver = new IntersectionObserver(
   379→    ([entry]) => {
   380→      sticky.classList.toggle("is-stuck", !entry.isIntersecting);
   381→    },
   382→    {
   383→      root: scrollContainer,
   384→      threshold: 0,
   385→      rootMargin: `-${QUESTION_BANK_STICKY_TOP}px 0px 0px 0px`,
   386→    },
   387→  );
   388→
   389→  questionBankStickyObserver.observe(sentinel);
   390→}

   488→function buildQuestionBankState(subject) {
   489→  const snapshot = getQuestionBankSnapshotBySubject(subject.id);
   490→  const query = getQuestionBankQuery(subject.id);
   491→  const entry = getQuestionBankEntryWithRecords(subject.id, snapshot.records || []);
   492→
   493→  if (subject.id !== "biology") {
   494→    return {
   495→      status: "ready",
   496→      query,
   497→      viewMode: entry.viewMode,
   498→      results: [],
   499→      filters: entry.filters,
   500→      options: {
   501→        sources: [],
   502→        chapters: [],
   503→        topics: [],
   504→        questionTypes: [],
   505→      },
   506→      hasActiveFilters: false,
   507→    };
   508→  }
   509→
   510→  if (snapshot.status !== "ready") {
   511→    return {
   512→      status: snapshot.status,
   513→      query,
   514→      viewMode: entry.viewMode,
   515→      results: [],
   516→      filters: entry.filters,
   517→      options: {
   518→        sources: QUESTION_BANK_SOURCE_OPTIONS,
   519→        chapters: [],
   520→        topics: [],
   521→        questionTypes: [],
   522→      },
   523→      hasActiveFilters: false,
   524→    };
   525→  }
   526→
   527→  const filters = normalizeQuestionBankFilters(snapshot.records, entry.filters);
   528→  entry.filters = filters;
   529→
   530→  return {
   531→    status: snapshot.status,
   532→    query,
   533→    viewMode: entry.viewMode,
   534→    filters,
   535→    options: buildQuestionBankFilterOptions(snapshot.records, filters.chapterIds),
   536→    hasActiveFilters: hasActiveQuestionBankFilters(snapshot.records, filters),
   537→    results: applyQuestionBankFilters(snapshot.records, { query, filters }),
   538→  };
   539→}

   716→function bindQuestionSearchShell(subjectId) {
   717→  const shell = document.querySelector("[data-question-search-shell]");
   718→  const input = document.querySelector("[data-question-search-input]");
   719→  if (!shell || !input) {
   720→    return { shell: null, input: null };
   721→  }
   722→
   723→  const entry = getQuestionBankStateEntry(subjectId);
   724→
   725→  const syncState = () => {
   726→    const hasValue = input.value.trim().length > 0;
   727→    shell.classList.toggle("has-value", hasValue);
   728→    if (!hasValue && document.activeElement !== input) {
   729→      shell.classList.remove("is-expanded");
   730→    }
   731→    entry.expandedSearch = shell.classList.contains("is-expanded") || hasValue;
   732→    return hasValue;
   733→  };
   734→
   735→  const expand = () => {
   736→    shell.classList.add("is-expanded");
   737→  };
   738→
   739→  if (getQuestionBankQuery(subjectId).trim()) {
   740→    shell.classList.add("is-expanded");
   741→    shell.classList.add("has-value");
   742→  } else if (entry.pendingRestore && entry.expandedSearch) {
   743→    shell.classList.add("is-expanded");
   744→  }
   745→
   746→  shell.addEventListener("pointerdown", (event) => {
   747→    if (event.target !== input) {
   748→      event.preventDefault();
   749→    }
   750→    expand();
   751→    setTimeout(() => input.focus(), 10);
   752→  });
   753→
   754→  input.addEventListener("focus", expand);
   755→  input.addEventListener("input", syncState);
   756→  input.addEventListener("blur", () => {
   757→    setTimeout(syncState, 90);
   758→  });
   759→
   760→  syncState();
   761→  return { shell, input };
   762→}

   773→function toggleFilterSelection(selectedIds, optionId, allowedIds) {
   774→  const current = Array.isArray(selectedIds) ? [...selectedIds] : [];
   775→  const filteredCurrent = current.filter((id) => allowedIds.includes(id));
   776→  const isSelected = filteredCurrent.includes(optionId);
   777→
   778→  if (isSelected) {
   779→    if (filteredCurrent.length === 1) {
   780→      return filteredCurrent;
   781→    }
   782→
   783→    return filteredCurrent.filter((id) => id !== optionId);
   784→  }
   785→
   786→  return [...filteredCurrent, optionId];
   787→}

   789→function bindQuestionBankInteractions(view) {
   790→  const subject = getQuestionBankSubjectByView(view);
   791→  if (!subject) {
   792→    return;
   793→  }
   794→
   795→  bindQuestionBankStickyState();
   796→  const { input } = bindQuestionSearchShell(subject.id);
   797→  const resultsContainer = document.querySelector("[data-question-bank-results]");
   798→
   799→  if (!resultsContainer) {
   800→    return;
   801→  }
   802→
   803→  const renderResults = () => {
   804→    const state = buildQuestionBankState(subject);
   805→    resultsContainer.innerHTML = renderQuestionBankResults(subject, state);
   806→  };
   807→
   808→  if (input) {
   809→    input.addEventListener("input", () => {
   810→      setQuestionBankQuery(subject.id, input.value);
   811→      renderResults();
   812→    });
   813→  }
   814→
   815→  resultsContainer.addEventListener("click", (event) => {
   816→    const resetButton = event.target.closest("[data-question-bank-reset-filters]");
   817→    if (resetButton && subject.id === "biology") {
   818→      updateQuestionBankView(subject.id, (entry) => {
   819→        const snapshot = getQuestionBankSnapshotBySubject(subject.id);
   820→        if (snapshot.status === "ready") {
   821→          entry.filters = createDefaultQuestionBankFilters(snapshot.records);
   822→        }
   823→        entry.isFilterPanelOpen = false;
   824→        entry.openAccordion = null;
   825→      });
   826→      return;
   827→    }
   828→
   829→    const button = event.target.closest("[data-question-detail-view]");
   830→    if (!button) {
   831→      return;
   832→    }
   833→
   834→    const detailView = button.getAttribute("data-question-detail-view");
   835→    if (detailView) {
   836→      saveQuestionBankViewState(subject.id, {
   837→        lastOpenedQuestionView: detailView,
   838→        isFilterPanelOpen: false,
   839→        openAccordion: null,
   840→      });
   841→      renderView(detailView, "tools");
   842→    }
   843→  });
   844→
   845→  const menuToggle = document.querySelector("[data-question-bank-menu-toggle]");
   846→  const panelDismiss = document.querySelector("[data-question-bank-panel-dismiss]");
   847→  const menuPanel = document.querySelector(".question-bank-menu-panel");
   848→
   849→  if (menuToggle && subject.id === "biology") {
   850→    menuToggle.addEventListener("click", () => {
   851→      updateQuestionBankView(subject.id, (entry) => {
   852→        entry.isFilterPanelOpen = !entry.isFilterPanelOpen;
   853→        if (!entry.isFilterPanelOpen) {
   854→          entry.openAccordion = null;
   855→        }
   856→      });
   857→    });
   858→  }
   859→
   860→  if (panelDismiss && subject.id === "biology") {
   861→    panelDismiss.addEventListener("click", () => {
   862→      updateQuestionBankView(subject.id, (entry) => {
   863→        entry.isFilterPanelOpen = false;
   864→        entry.openAccordion = null;
   865→      });
   866→    });
   867→  }
   868→
   869→  if (menuPanel && subject.id === "biology") {
   870→    menuPanel.addEventListener("click", (event) => {
   871→      const accordionButton = event.target.closest("[data-question-bank-accordion]");
   872→      if (accordionButton) {
   873→        const group = accordionButton.getAttribute("data-question-bank-accordion");
   874→        updateQuestionBankView(subject.id, (entry) => {
   875→          entry.isFilterPanelOpen = true;
   876→          entry.openAccordion = entry.openAccordion === group ? null : group;
   877→        });
   878→        return;
   879→      }
   880→
   881→      const viewModeButton = event.target.closest("[data-question-bank-view-mode]");
   882→      if (viewModeButton) {
   883→        const nextMode = viewModeButton.getAttribute("data-question-bank-view-mode");
   884→        updateQuestionBankView(subject.id, (entry) => {
   885→          entry.isFilterPanelOpen = true;
   886→          entry.viewMode = nextMode === "icons" ? "icons" : "list";
   887→        });
   888→        return;
   889→      }
   890→
   891→      const resetButton = event.target.closest("[data-question-bank-reset-filters]");
   892→      if (resetButton) {
   893→        updateQuestionBankView(subject.id, (entry) => {
   894→          const snapshot = getQuestionBankSnapshotBySubject(subject.id);
   895→          if (snapshot.status === "ready") {
   896→            entry.filters = createDefaultQuestionBankFilters(snapshot.records);
   897→          }
   898→          entry.isFilterPanelOpen = true;
   899→          entry.openAccordion = null;
   900→        });
   901→        return;
   902→      }
   903→
   904→      const optionButton = event.target.closest("[data-filter-group][data-filter-option]");
   905→      if (!optionButton) {
   906→        return;
   907→      }
   908→
   909→      const group = optionButton.getAttribute("data-filter-group");
   910→      const optionId = optionButton.getAttribute("data-filter-option");
   911→
   912→      updateQuestionBankView(subject.id, (entry) => {
   913→        const snapshot = getQuestionBankSnapshotBySubject(subject.id);
   914→        if (snapshot.status !== "ready") {
   915→          return;
   916→        }
   917→
   918→        const currentFilters = normalizeQuestionBankFilters(snapshot.records, entry.filters || {});
   919→        const currentOptions = buildQuestionBankFilterOptions(snapshot.records, currentFilters.chapterIds);
   920→        const optionMap = {
   921→          sourceIds: currentOptions.sources.map((item) => item.id),
   922→          chapterIds: currentOptions.chapters.map((item) => item.id),
   923→          topicIds: currentOptions.topics.map((item) => item.id),
   924→          questionTypeIds: currentOptions.questionTypes.map((item) => item.id),
   925→        };
   926→
   927→        if (!optionMap[group]) {
   928→          return;
   929→        }
   930→
   931→        currentFilters[group] = toggleFilterSelection(currentFilters[group], optionId, optionMap[group]);
   932→
   933→        if (group === "chapterIds") {
   934→          currentFilters.topicIds = normalizeQuestionBankFilters(snapshot.records, currentFilters).topicIds;
   935→        }
   936→
   937→        entry.filters = normalizeQuestionBankFilters(snapshot.records, currentFilters);
   938→        entry.isFilterPanelOpen = true;
   939→        entry.openAccordion = group;
   940→      });
   941→    });
   942→  }
   943→
   944→  const entry = getQuestionBankStateEntry(subject.id);
   945→  if (entry.isFilterPanelOpen) {
   946→    const handler = (event) => {
   947→      if (event.target.closest(".question-bank-floating-tools")) {
   948→        return;
   949→      }
   950→
   951→      cleanupQuestionBankOutsidePointerListener();
   952→      updateQuestionBankView(subject.id, (nextEntry) => {
   953→        nextEntry.isFilterPanelOpen = false;
   954→        nextEntry.openAccordion = null;
   955→      });
   956→    };
   957→
   958→    document.addEventListener("pointerdown", handler, true);
   959→    questionBankOutsidePointerCleanup = () => {
   960→      document.removeEventListener("pointerdown", handler, true);
   961→    };
   962→  }
   963→
   964→  if (subject.id === "biology") {
   965→    ensureBiologyQuestionBankLoaded();
   966→  }
   967→
   968→  const snapshot = buildQuestionBankState(subject);
   969→  if (snapshot.status === "ready" || subject.id !== "biology") {
   970→    restoreQuestionBankViewState(subject.id, { allowQuestionFallback: true });
   971→  }
   972→}
