   158→function restoreQuestionBankViewState(subjectId, options = {}) {
   159→  const entry = getQuestionBankStateEntry(subjectId);
   160→  if (!entry.pendingRestore) {
   161→    return;
   162→  }
   163→
   164→  const scrollContainer = getScreenScrollContainer();
   165→  if (!scrollContainer) {
   166→    return;
   167→  }
   168→
   169→  const restore = () => {
   170→    const targetScrollTop = Math.max(0, entry.scrollTop || 0);
   171→    scrollContainer.scrollTop = targetScrollTop;
   172→
   173→    if (options.allowQuestionFallback && entry.useQuestionFallbackOnRestore && entry.lastOpenedQuestionView) {
   174→      const questionCard = document.querySelector(`[data-question-detail-view="${entry.lastOpenedQuestionView}"]`);
   175→      const delta = Math.abs((scrollContainer.scrollTop || 0) - targetScrollTop);
   176→      if (questionCard && delta > 120) {
   177→        questionCard.scrollIntoView({ block: "center" });
   178→      }
   179→    }
   180→
   181→    entry.pendingRestore = false;
   182→    entry.useQuestionFallbackOnRestore = false;
   183→  };
   184→
   185→  requestAnimationFrame(() => {
   186→    requestAnimationFrame(restore);
   187→  });

   229→function summarizeSelectedOptions(selectedIds, options) {
   230→  const available = Array.isArray(options) ? options : [];
   231→  const selected = available.filter((option) => selectedIds.includes(option.id));
   232→
   233→  if (!available.length || selected.length === available.length) {
   234→    return "الكل";
   235→  }
   236→
   237→  if (selected.length === 1) {
   238→    return selected[0].label;
   239→  }
   240→
   241→  if (selected.length === 2) {
   242→    return `${selected[0].label} + ${selected[1].label}`;
   243→  }
   244→
   245→  return `${selected.length} محددة`;
   246→}

   375→function bindQuestionBankStickyState() {
   376→  cleanupQuestionBankStickyObserver();
   377→
   378→  const sticky = document.querySelector("[data-question-search-sticky]");
   379→  const sentinel = document.querySelector("[data-question-search-sentinel]");
   380→  const scrollContainer = getScreenScrollContainer();
   381→
   382→  if (!sticky || !sentinel || !scrollContainer || !("IntersectionObserver" in window)) {
   383→    return;
   384→  }
   385→
   386→  questionBankStickyObserver = new IntersectionObserver(
   387→    ([entry]) => {
   388→      sticky.classList.toggle("is-stuck", !entry.isIntersecting);
   389→    },
   390→    {
   391→      root: scrollContainer,
   392→      threshold: 0,
   393→      rootMargin: `-${QUESTION_BANK_STICKY_TOP}px 0px 0px 0px`,
   394→    },
   395→  );
   396→
   397→  questionBankStickyObserver.observe(sentinel);
   398→}

   496→function buildQuestionBankState(subject) {
   497→  const snapshot = getQuestionBankSnapshotBySubject(subject.id);
   498→  const query = getQuestionBankQuery(subject.id);
   499→  const entry = getQuestionBankEntryWithRecords(subject.id, snapshot.records || []);
   500→
   501→  if (subject.id !== "biology") {
   502→    return {
   503→      status: "ready",
   504→      query,
   505→      viewMode: entry.viewMode,
   506→      results: [],
   507→      filters: entry.filters,
   508→      options: {
   509→        sources: [],
   510→        chapters: [],
   511→        topics: [],
   512→        questionTypes: [],
   513→      },
   514→      hasActiveFilters: false,
   515→    };
   516→  }
   517→
   518→  if (snapshot.status !== "ready") {
   519→    return {
   520→      status: snapshot.status,
   521→      query,
   522→      viewMode: entry.viewMode,
   523→      results: [],
   524→      filters: entry.filters,
   525→      options: {
   526→        sources: QUESTION_BANK_SOURCE_OPTIONS,
   527→        chapters: [],
   528→        topics: [],
   529→        questionTypes: [],
   530→      },
   531→      hasActiveFilters: false,
   532→    };
   533→  }
   534→
   535→  const filters = normalizeQuestionBankFilters(snapshot.records, entry.filters);
   536→  entry.filters = filters;
   537→
   538→  return {
   539→    status: snapshot.status,
   540→    query,
   541→    viewMode: entry.viewMode,
   542→    filters,
   543→    options: buildQuestionBankFilterOptions(snapshot.records, filters.chapterIds),
   544→    hasActiveFilters: hasActiveQuestionBankFilters(snapshot.records, filters),
   545→    results: applyQuestionBankFilters(snapshot.records, { query, filters }),
   546→  };
   547→}

   566→

   569→  }
   570→
   571→  if (view === "tests") {
   572→    return testsSubjectsScreen();
   573→  }
   574→
   575→  if (biologyQuestionGlobal !== null) {
   576→    return questionDetailScreen(biologyQuestionGlobal, buildBiologyDetailState(biologyQuestionGlobal));

   578→
   579→  if (questionBankSubject) {

   582→
   583→  if (subject) {
   584→    return subjectTestsScreen(subject);
   585→  }
   586→
   587→  if (view === "settings") {
   588→    return settingsScreen();
   589→  }

   758→function bindQuestionSearchShell(subjectId) {
   759→  const shell = document.querySelector("[data-question-search-shell]");
   760→  const input = document.querySelector("[data-question-search-input]");
   761→  if (!shell || !input) {
   762→    return { shell: null, input: null };
   763→  }
   764→
   765→  const entry = getQuestionBankStateEntry(subjectId);
   766→
   767→  const syncState = () => {
   768→    const hasValue = input.value.trim().length > 0;
   769→    shell.classList.toggle("has-value", hasValue);
   770→    if (!hasValue && document.activeElement !== input) {
   771→      shell.classList.remove("is-expanded");
   772→    }
   773→    entry.expandedSearch = shell.classList.contains("is-expanded") || hasValue;
   774→    return hasValue;
   775→  };
   776→
   777→  const expand = () => {
   778→    shell.classList.add("is-expanded");
   779→  };
   780→
   781→  if (getQuestionBankQuery(subjectId).trim()) {
   782→    shell.classList.add("is-expanded");
   783→    shell.classList.add("has-value");
   784→  } else if (entry.pendingRestore && entry.expandedSearch) {
   785→    shell.classList.add("is-expanded");
   786→  }
   787→
   788→  shell.addEventListener("pointerdown", (event) => {
   789→    if (event.target !== input) {
   790→      event.preventDefault();
   791→    }
   792→    expand();
   793→    setTimeout(() => input.focus(), 10);
   794→  });
   795→
   796→  input.addEventListener("focus", expand);
   797→  input.addEventListener("input", syncState);
   798→  input.addEventListener("blur", () => {
   799→    setTimeout(syncState, 90);
   800→  });
   801→
   802→  syncState();
   803→  return { shell, input };
   804→}

   815→function toggleFilterSelection(selectedIds, optionId, allowedIds) {
   816→  const current = Array.isArray(selectedIds) ? [...selectedIds] : [];
   817→  const filteredCurrent = current.filter((id) => allowedIds.includes(id));
   818→  const isSelected = filteredCurrent.includes(optionId);
   819→
   820→  if (isSelected) {
   821→    if (filteredCurrent.length === 1) {
   822→      return filteredCurrent;
   823→    }
   824→
   825→    return filteredCurrent.filter((id) => id !== optionId);
   826→  }
   827→
   828→  return [...filteredCurrent, optionId];
   829→}

   831→function bindQuestionBankInteractions(view) {
   832→  const subject = getQuestionBankSubjectByView(view);
   833→  if (!subject) {
   834→    return;
   835→  }
   836→
   837→  bindQuestionBankStickyState();
   838→  const { input } = bindQuestionSearchShell(subject.id);
   839→  const resultsContainer = document.querySelector("[data-question-bank-results]");
   840→
   841→  if (!resultsContainer) {
   842→    return;
   843→  }
   844→
   845→  const renderResults = () => {
   846→    const state = buildQuestionBankState(subject);
   847→    resultsContainer.innerHTML = renderQuestionBankResults(subject, state);
   848→  };
   849→
   850→  if (input) {
   851→    input.addEventListener("input", () => {
   852→      setQuestionBankQuery(subject.id, input.value);
   853→      renderResults();
   854→    });
   855→  }
   856→
   857→  resultsContainer.addEventListener("click", (event) => {
   858→    const resetButton = event.target.closest("[data-question-bank-reset-filters]");
   859→    if (resetButton && subject.id === "biology") {
   860→      updateQuestionBankView(subject.id, (entry) => {
   861→        const snapshot = getQuestionBankSnapshotBySubject(subject.id);
   862→        if (snapshot.status === "ready") {
   863→          entry.filters = createDefaultQuestionBankFilters(snapshot.records);
   864→        }
   865→        entry.isFilterPanelOpen = false;
   866→        entry.openAccordion = null;
   867→      });
   868→      return;
   869→    }
   870→
   871→    const button = event.target.closest("[data-question-detail-view]");
   872→    if (!button) {
   873→      return;
   874→    }
   875→
   876→    const detailView = button.getAttribute("data-question-detail-view");
   877→    if (detailView) {
   878→      saveQuestionBankViewState(subject.id, {
   879→        lastOpenedQuestionView: detailView,
   880→        isFilterPanelOpen: false,
   881→        openAccordion: null,
   882→      });
   883→      renderView(detailView, "tools");
   884→    }
   885→  });
   886→
   887→  const menuToggle = document.querySelector("[data-question-bank-menu-toggle]");
   888→  const panelDismiss = document.querySelector("[data-question-bank-panel-dismiss]");
   889→  const menuPanel = document.querySelector(".question-bank-menu-panel");
   890→
   891→  if (menuToggle && subject.id === "biology") {
   892→    menuToggle.addEventListener("click", () => {
   893→      updateQuestionBankView(subject.id, (entry) => {
   894→        entry.isFilterPanelOpen = !entry.isFilterPanelOpen;
   895→        if (!entry.isFilterPanelOpen) {
   896→          entry.openAccordion = null;
   897→        }
   898→      });
   899→    });
   900→  }
   901→
   902→  if (panelDismiss && subject.id === "biology") {
   903→    panelDismiss.addEventListener("click", () => {
   904→      updateQuestionBankView(subject.id, (entry) => {
   905→        entry.isFilterPanelOpen = false;
   906→        entry.openAccordion = null;
   907→      });
   908→    });
   909→  }
   910→
   911→  if (menuPanel && subject.id === "biology") {
   912→    menuPanel.addEventListener("click", (event) => {
   913→      const accordionButton = event.target.closest("[data-question-bank-accordion]");
   914→      if (accordionButton) {
   915→        const group = accordionButton.getAttribute("data-question-bank-accordion");
   916→        updateQuestionBankView(subject.id, (entry) => {
   917→          entry.isFilterPanelOpen = true;
   918→          entry.openAccordion = entry.openAccordion === group ? null : group;
   919→        });
   920→        return;
   921→      }
   922→
   923→      const viewModeButton = event.target.closest("[data-question-bank-view-mode]");
   924→      if (viewModeButton) {
   925→        const nextMode = viewModeButton.getAttribute("data-question-bank-view-mode");
   926→        updateQuestionBankView(subject.id, (entry) => {
   927→          entry.isFilterPanelOpen = true;
   928→          entry.viewMode = nextMode === "icons" ? "icons" : "list";
   929→        });
   930→        return;
   931→      }
   932→
   933→      const resetButton = event.target.closest("[data-question-bank-reset-filters]");
   934→      if (resetButton) {
   935→        updateQuestionBankView(subject.id, (entry) => {
   936→          const snapshot = getQuestionBankSnapshotBySubject(subject.id);
   937→          if (snapshot.status === "ready") {
   938→            entry.filters = createDefaultQuestionBankFilters(snapshot.records);
   939→          }
   940→          entry.isFilterPanelOpen = true;
   941→          entry.openAccordion = null;
   942→        });
   943→        return;
   944→      }
   945→
   946→      const optionButton = event.target.closest("[data-filter-group][data-filter-option]");
   947→      if (!optionButton) {
   948→        return;
   949→      }
   950→
   951→      const group = optionButton.getAttribute("data-filter-group");
   952→      const optionId = optionButton.getAttribute("data-filter-option");
   953→
   954→      updateQuestionBankView(subject.id, (entry) => {
   955→        const snapshot = getQuestionBankSnapshotBySubject(subject.id);
   956→        if (snapshot.status !== "ready") {
   957→          return;
   958→        }
   959→
   960→        const currentFilters = normalizeQuestionBankFilters(snapshot.records, entry.filters || {});
   961→        const currentOptions = buildQuestionBankFilterOptions(snapshot.records, currentFilters.chapterIds);
   962→        const optionMap = {
   963→          sourceIds: currentOptions.sources.map((item) => item.id),
   964→          chapterIds: currentOptions.chapters.map((item) => item.id),
   965→          topicIds: currentOptions.topics.map((item) => item.id),
   966→          questionTypeIds: currentOptions.questionTypes.map((item) => item.id),
   967→        };
   968→
   969→        if (!optionMap[group]) {
   970→          return;
   971→        }
   972→
   973→        currentFilters[group] = toggleFilterSelection(currentFilters[group], optionId, optionMap[group]);
   974→
   975→        if (group === "chapterIds") {
   976→          currentFilters.topicIds = normalizeQuestionBankFilters(snapshot.records, currentFilters).topicIds;
   977→        }
   978→
   979→        entry.filters = normalizeQuestionBankFilters(snapshot.records, currentFilters);
   980→        entry.isFilterPanelOpen = true;
   981→        entry.openAccordion = group;
   982→      });
   983→    });
   984→  }
   985→
   986→  const entry = getQuestionBankStateEntry(subject.id);
   987→  if (entry.isFilterPanelOpen) {
   988→    const handler = (event) => {
   989→      if (event.target.closest(".question-bank-floating-tools")) {
   990→        return;
   991→      }
   992→
   993→      cleanupQuestionBankOutsidePointerListener();
   994→      updateQuestionBankView(subject.id, (nextEntry) => {
   995→        nextEntry.isFilterPanelOpen = false;
   996→        nextEntry.openAccordion = null;
   997→      });
   998→    };
   999→
  1000→    document.addEventListener("pointerdown", handler, true);
  1001→    questionBankOutsidePointerCleanup = () => {
  1002→      document.removeEventListener("pointerdown", handler, true);
  1003→    };
  1004→  }
  1005→
  1006→  if (subject.id === "biology") {
  1007→    ensureBiologyQuestionBankLoaded();
  1008→  }
  1009→
  1010→  const snapshot = buildQuestionBankState(subject);
  1011→  if (snapshot.status === "ready" || subject.id !== "biology") {
  1012→    restoreQuestionBankViewState(subject.id, { allowQuestionFallback: true });
  1013→  }
  1014→}
