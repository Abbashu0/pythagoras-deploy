   758→   758→function bindQuestionSearchShell(subjectId) {
   759→   759→  const shell = document.querySelector("[data-question-search-shell]");
   760→   760→  const input = document.querySelector("[data-question-search-input]");
   761→   761→  if (!shell || !input) {
   762→   762→    return { shell: null, input: null };
   763→   763→  }
   764→   764→
   765→   765→  const entry = getQuestionBankStateEntry(subjectId);
   766→   766→
   767→   767→  const syncState = () => {
   768→   768→    const hasValue = input.value.trim().length > 0;
   769→   769→    shell.classList.toggle("has-value", hasValue);
   770→   770→    if (!hasValue && document.activeElement !== input) {
   771→   771→      shell.classList.remove("is-expanded");
   772→   772→    }
   773→   773→    entry.expandedSearch = shell.classList.contains("is-expanded") || hasValue;
   774→   774→    return hasValue;
   775→   775→  };
   776→   776→
   777→   777→  const expand = () => {
   778→   778→    shell.classList.add("is-expanded");
   779→   779→  };
   780→   780→
   781→   781→  if (getQuestionBankQuery(subjectId).trim()) {
   782→   782→    shell.classList.add("is-expanded");
   783→   783→    shell.classList.add("has-value");
   784→   784→  } else if (entry.pendingRestore && entry.expandedSearch) {
   785→   785→    shell.classList.add("is-expanded");
   786→   786→  }
   787→   787→
   788→   788→  shell.addEventListener("pointerdown", (event) => {
   789→   789→    if (event.target !== input) {
   790→   790→      event.preventDefault();
   791→   791→    }
   792→   792→    expand();
   793→   793→    setTimeout(() => input.focus(), 10);
   794→   794→  });
   795→   795→
   796→   796→  input.addEventListener("focus", expand);
   797→   797→  input.addEventListener("input", syncState);
   798→   798→  input.addEventListener("blur", () => {
   799→   799→    setTimeout(syncState, 90);
   800→   800→  });
   801→   801→
   802→   802→  syncState();
   803→   803→  return { shell, input };
   804→   804→}

   815→   815→function toggleFilterSelection(selectedIds, optionId, allowedIds) {
   816→   816→  const current = Array.isArray(selectedIds) ? [...selectedIds] : [];
   817→   817→  const filteredCurrent = current.filter((id) => allowedIds.includes(id));
   818→   818→  const isSelected = filteredCurrent.includes(optionId);
   819→   819→
   820→   820→  if (isSelected) {
   821→   821→    if (filteredCurrent.length === 1) {
   822→   822→      return filteredCurrent;
   823→   823→    }
   824→   824→
   825→   825→    return filteredCurrent.filter((id) => id !== optionId);
   826→   826→  }
   827→   827→
   828→   828→  return [...filteredCurrent, optionId];
   829→   829→}
