     1→"use client";
     2→
     3→/**
     4→ * AdminStore — the single source of truth for the Sponsored Carousel
     5→ * admin dashboard.
     6→ *
     7→ * Responsibilities:
     8→ *   - Hold the list of SponsoredBanner objects (max 5).
     9→ *   - Hold the ActivityHistoryEntry log (newest first, cap 200).
    10→ *   - Hold admin theme ("light" | "dark").
    11→ *   - Hold carousel autoSlideInterval (ms, default 10000).
    12→ *   - Persist all four to localStorage so reloads don't lose work.
    13→ *   - Surface QuotaExceededError via `lastStorageError` so the UI can warn.
    14→ *   - Emit change events so React components can subscribe via
    15→ *     `useSyncExternalStore`.
    16→ *
    17→ * Architecture notes:
    18→ *   - This is a plain JS class (not React state) so business logic stays
    19→ *     out of UI components. Components subscribe via `useSyncExternalStore`
    20→ *     (see `useAdminStore` hook).
    21→ *   - When the backend arrives, swap the localStorage persistence layer
    22→ *     for API calls. The public method signatures (add, update, delete,
    23→ *     move, duplicate, setAdminTheme, setAutoSlideInterval) stay the same,
    24→ *     so UI components won't change.
    25→ *   - The store singleton is created during SSR (where there's no
    26→ *     localStorage) AND on the client during hydration. To avoid hydration
    27→ *     mismatches, the constructor does NOT load from localStorage — instead
    28→ *     `loadFromStorage()` is called from a `useEffect` in the admin page
    29→ *     (client-only, after hydration).
    30→ */
    31→

    49→const HISTORY_KEY = "pythagoras-admin-history";
    50→const THEME_KEY = "pythagoras-admin-theme";
    51→const CAROUSEL_KEY = "pythagoras-admin-carousel-settings";

    57→/**
    58→ * The immutable snapshot returned by `useAdminStore()`.
    59→ *
    60→ * Components read fields off this object; they never mutate it directly.
    61→ * The reference is cached on the store and only changes when state
    62→ * actually changes (so `useSyncExternalStore` doesn't loop forever).
    63→ */
    64→export interface AdminSnapshot {
    65→  banners: SponsoredBanner[];
    66→  history: ActivityHistoryEntry[];
    67→  canAddMore: boolean;
    68→  adminTheme: "light" | "dark";
    69→  autoSlideInterval: number;
    70→  /** Last quota/storage error message, or null if everything is fine. */
    71→  lastStorageError: string | null;
    72→}
    73→
    74→type Listener = () => void;
    75→
    76→class AdminStore {
    77→  private banners: SponsoredBanner[] = [];
    78→  private history: ActivityHistoryEntry[] = [];
    79→  private listeners: Set<Listener> = new Set();
    80→  private hasLoadedFromStorage = false;
    81→
    82→  adminTheme: "light" | "dark" = DEFAULT_ADMIN_THEME;
    83→  autoSlideInterval: number = DEFAULT_AUTO_SLIDE_INTERVAL;
    84→
    85→  /**
    86→   * Last quota error message, or null if no error.
    87→   * The admin UI reads this to show a warning banner when localStorage
    88→   * is full.
    89→   */
    90→  lastStorageError: string | null = null;
    91→
    92→  // Cached snapshot — useSyncExternalStore requires getSnapshot to return
    93→  // a stable reference when nothing has changed, otherwise it loops
    94→  // forever. We rebuild this object only when state changes.
    95→  private snapshot: AdminSnapshot = {
    96→    banners: [],
    97→    history: [],
    98→    canAddMore: true,
    99→    adminTheme: DEFAULT_ADMIN_THEME,
   100→    autoSlideInterval: DEFAULT_AUTO_SLIDE_INTERVAL,
   101→    lastStorageError: null,
   102→  };
   103→
   104→  constructor() {
   105→    // IMPORTANT: do NOT load from localStorage in the constructor.
   106→    // The store singleton is created during SSR (where there's no
   107→    // localStorage) AND on the client during hydration. If we load here,
   108→    // the server snapshot returns 0 banners while the client returns 5 →
   109→    // hydration mismatch.
   110→    //
   111→    // Instead, we start with an empty snapshot (0 banners, default theme,
   112→    // default interval) on BOTH sides so the initial render matches, then
   113→    // call `loadFromStorage()` from a `useEffect` in the admin page
   114→    // (client-only, after hydration).
   115→    this.rebuildSnapshot();
   116→  }
   117→
   118→  /**
   119→   * Load banners + history + theme + carousel settings from localStorage.
   120→   * Called once on the client AFTER hydration completes (via useEffect in
   121→   * the admin page).
   122→   *
   123→   * Returns true if banners were loaded (or seeded), false if storage was
   124→   * empty and we didn't seed (shouldn't happen — we always seed on first
   125→   * visit).
   126→   */
   127→  loadFromStorage(): boolean {
   128→    if (this.hasLoadedFromStorage) return this.banners.length > 0;
   129→    this.hasLoadedFromStorage = true;
   130→    if (typeof window === "undefined") return false;
   131→    try {
   132→      // 1. Theme
   133→      const t = localStorage.getItem(THEME_KEY);
   134→      if (t === "light" || t === "dark") {
   135→        this.adminTheme = t;
   136→      }
   137→
   138→      // 2. Carousel settings
   139→      const c = localStorage.getItem(CAROUSEL_KEY);
   140→      if (c) {
   141→        try {
   142→          const parsed = JSON.parse(c) as { autoSlideInterval?: unknown };
   143→          if (
   144→            parsed &&
   145→            typeof parsed.autoSlideInterval === "number" &&
   146→            Number.isFinite(parsed.autoSlideInterval) &&
   147→            parsed.autoSlideInterval > 0
   148→          ) {
   149→            this.autoSlideInterval = parsed.autoSlideInterval;
   150→          }
   151→        } catch {
   152→          /* ignore malformed carousel settings */
   153→        }
   154→      }
   155→
   156→      // 3. Banners (with migration for older shapes)
   157→      const b = localStorage.getItem(BANNERS_KEY);
   158→      if (b) {
   159→        const parsed = JSON.parse(b) as SponsoredBanner[];
   160→        // SANITIZER: drop any large `image` data URL field — it lives
   161→        // in IndexedDB now (via imageKey) and would blow localStorage's
   162→        // ~5MB quota. We keep ONLY the imageKey reference.
   163→        // HYDRATION: if the banner has an imageKey, resolve it to the
   164→        // actual data URL via ImageDB.getImageSync() so the in-memory
   165→        // banner object can be rendered by AdminBannerCard,
   166→        // LiveCarouselPreview, etc.
   167→        this.banners = parsed.map((banner) => {
   168→          const sanitized = { ...banner };
   169→          if (
   170→            typeof sanitized.image === "string" &&
   171→            sanitized.image.startsWith("data:") &&
   172→            sanitized.image.length > 1000
   173→          ) {
   174→            sanitized.image = "";
   175→          }
   176→          // Hydrate from IndexedDB cache if available (synchronous read
   177→          // from the in-memory Map; returns "" if not yet preloaded).
   178→          if (sanitized.imageKey && !sanitized.image) {
   179→            const dataUrl = getImageSync(sanitized.imageKey);
   180→            if (dataUrl) sanitized.image = dataUrl;
   181→          }
   182→          return {
   183→            ...sanitized,
   184→            bannerType: (sanitized.bannerType ?? "split") as BannerType,
   185→            transform: sanitized.transform || { ...BANNER_TRANSFORM_DEFAULT },
   186→          };
   187→        });
   188→        this.banners.sort((a, b2) => a.displayOrder - b2.displayOrder);
   189→      } else {
   190→        // First visit — seed with the default 5 banners so the dashboard
   191→        // isn't empty on first load.
   192→        this.banners = this.seedBanners();
   193→      }
   194→
   195→      // 4. History — with one-time sanitizer.
   196→      // Older versions stored FULL image data URLs in each entry's
   197→      // `thumbnail` field. Strip oversized thumbnails to reclaim space.
   198→      const h = localStorage.getItem(HISTORY_KEY);
   199→      if (h) {
   200→        const parsed = JSON.parse(h) as ActivityHistoryEntry[];
   201→        this.history = parsed.map((entry) => {
   202→          if (
   203→            entry &&
   204→            typeof entry.thumbnail === "string" &&
   205→            entry.thumbnail.startsWith("data:") &&
   206→            entry.thumbnail.length > 1000
   207→          ) {
   208→            return { ...entry, thumbnail: "" };
   209→          }
   210→          return entry;
   211→        });
   212→      }
   213→
   214→      // Persist — writes migrated banner shapes back + ensures all four
   215→      // keys exist (useful when upgrading an older install that doesn't
   216→      // have theme/carousel keys yet).
   217→      this.persist();
   218→      this.emit();
   219→      return this.banners.length > 0;
   220→    } catch {
   221→      this.banners = this.seedBanners();
   222→      this.persist();
   223→      this.emit();
   224→      return true;
   225→    }
   226→  }
   227→
   228→  /**
   229→   * Asynchronously hydrate banner images from IndexedDB.
   230→   * Call after loadFromStorage() on admin pages so list cards + the
   231→   * editor + live preview can render uploaded images.
   232→   */
   233→  async hydrateImagesFromIDB(): Promise<void> {
   234→    if (typeof window === "undefined") return;
   235→    try {
   236→      await preloadAllImages();
   237→      let changed = false;
   238→      this.banners = this.banners.map((b) => {
   239→        if (b.imageKey && (!b.image || b.image.length === 0)) {
   240→          const dataUrl = getImageSync(b.imageKey);
   241→          if (dataUrl) {
   242→            changed = true;
   243→            return { ...b, image: dataUrl };
   244→          }
   245→        }
   246→        return b;
   247→      });
   248→      if (changed) this.emit();
   249→    } catch (e) {
   250→      console.warn("[AdminStore] hydrateImagesFromIDB failed:", e);
   251→    }
   252→  }

   319→  /**
   320→   * Thumbnail for a banner — used in activity log entries.
   321→   * Returns the image data URL when an image is uploaded, otherwise
   322→   * falls back to the gradient string (the UI can use it as a CSS
   323→   * background).
   324→   */
   325→  bannerThumbnail(banner: SponsoredBanner): string {
   326→    // Return `idb:${imageKey}` (a short reference string) so the
   327→    // ActivityHistory panel can resolve it via ImageDB.getImageSync()
   328→    // at render time. Falls back to the gradient (a short CSS string).
   329→    if (banner.imageKey) return `idb:${banner.imageKey}`;
   330→    const img = banner.image || "";
   331→    if (img.startsWith("data:")) return ""; // legacy data URL — drop
   332→    if (img) return img;
   333→    return banner.gradient || "";
   334→  }

   336→  // ---------- Subscription ----------
   337→  subscribe = (listener: Listener): (() => void) => {
   338→    this.listeners.add(listener);
   339→    return () => {
   340→      this.listeners.delete(listener);
   341→    };
   342→  };
   343→
   344→  private emit() {
   345→    this.rebuildSnapshot();
   346→    this.listeners.forEach((l) => l());
   347→  }
   348→
   349→  private rebuildSnapshot() {
   350→    this.snapshot = {
   351→      banners: this.banners,
   352→      history: this.history,
   353→      canAddMore: this.getActiveBanners().length < MAX_ACTIVE_BANNERS,
   354→      adminTheme: this.adminTheme,
   355→      autoSlideInterval: this.autoSlideInterval,
   356→      lastStorageError: this.lastStorageError,
   357→    };
   358→  }
   359→
   360→  getSnapshot = (): AdminSnapshot => {
   361→    return this.snapshot;
   362→  };
   363→
   364→  getServerSnapshot = (): AdminSnapshot => {
   365→    // Server has no localStorage — return the cached snapshot (which on
   366→    // the server is always the initial empty/default snapshot since
   367→    // loadFromStorage is never called server-side).
   368→    return this.snapshot;
   369→  };

   371→  // ---------- Persistence ----------
   372→  private persist() {
   373→    if (typeof window === "undefined") return;
   374→    try {
   375→      // CRITICAL: Strip image DATA URLs from banners before saving to
   376→      // localStorage. The image data lives in IndexedDB (via imageKey).
   377→      const bannersForStorage = this.banners.map((b) => {
   378→        if (b.imageKey) return { ...b, image: "" };
   379→        if (
   380→          typeof b.image === "string" &&
   381→          b.image.startsWith("data:") &&
   382→          b.image.length > 1000
   383→        ) {
   384→          return { ...b, image: "" };
   385→        }
   386→        return b;
   387→      });
   388→      localStorage.setItem(BANNERS_KEY, JSON.stringify(bannersForStorage));
   389→      localStorage.setItem(HISTORY_KEY, JSON.stringify(this.history));
   390→      localStorage.setItem(THEME_KEY, this.adminTheme);
   391→      localStorage.setItem(
   392→        CAROUSEL_KEY,
   393→        JSON.stringify({ autoSlideInterval: this.autoSlideInterval })
   394→      );
   395→      this.lastStorageError = null;
   396→    } catch (err) {
   397→      // QuotaExceededError — localStorage is full.
   398→      // Try recovery: clear keys + retry with sanitized data.
   399→      const isQuota =
   400→        err instanceof DOMException &&
   401→        (err.name === "QuotaExceededError" ||
   402→          err.name === "NS_ERROR_DOM_QUOTA_REACHED" ||
   403→          err.code === 22 ||
   404→          err.code === 1014);
   405→
   406→      if (isQuota) {
   407→        try {
   408→          localStorage.removeItem(BANNERS_KEY);
   409→          localStorage.removeItem(HISTORY_KEY);
   410→          const bannersForStorage = this.banners.map((b) => {
   411→            if (b.imageKey) return { ...b, image: "" };
   412→            if (
   413→              typeof b.image === "string" &&
   414→              b.image.startsWith("data:") &&
   415→              b.image.length > 1000
   416→            ) {
   417→              return { ...b, image: "" };
   418→            }
   419→            return b;
   420→          });
   421→          localStorage.setItem(BANNERS_KEY, JSON.stringify(bannersForStorage));
   422→          localStorage.setItem(HISTORY_KEY, JSON.stringify(this.history));
   423→          localStorage.setItem(THEME_KEY, this.adminTheme);
   424→          localStorage.setItem(
   425→            CAROUSEL_KEY,
   426→            JSON.stringify({ autoSlideInterval: this.autoSlideInterval })
   427→          );
   428→          this.lastStorageError = null;
   429→          console.info("[AdminStore] Recovery successful after quota error.");
   430→          return;
   431→        } catch (recoveryErr) {
   432→          console.error("[AdminStore] Recovery write also failed:", recoveryErr);
   433→        }
   434→      }
   435→
   436→      this.lastStorageError = isQuota
   437→        ? "امتلأت مساحة التخزين المحلية. احذف بانراً قديماً أو استخدم صوراً أصغر."
   438→        : `خطأ في الحفظ: ${err instanceof Error ? err.message : String(err)}`;
   439→      console.warn("[AdminStore] persist failed:", this.lastStorageError, err);
   440→    }
   441→  }

   443→  // ---------- Activity logging ----------
   444→  /**
   445→   * Prepend a new ActivityHistoryEntry to the history array.
   446→   * Cap at HISTORY_CAP entries (newest kept).
   447→   *
   448→   * NOTE: this method only mutates `this.history`. Callers are
   449→   * responsible for calling `persist()` and `emit()` afterwards so the
   450→   * change is saved and subscribers are notified.
   451→   */
   452→  private log(
   453→    action: ActivityAction,
   454→    bannerTitle: string,
   455→    thumbnail: string,
   456→    changeSummary?: string[]
   457→  ) {
   458→    const entry: ActivityHistoryEntry = {
   459→      id:
   460→        typeof crypto !== "undefined" && "randomUUID" in crypto
   461→          ? crypto.randomUUID()
   462→          : `log-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   463→      at: new Date().toISOString(),
   464→      action,
   465→      label: ACTIVITY_LABELS[action],
   466→      bannerTitle,
   467→      thumbnail,
   468→      changeSummary,
   469→    };
   470→    this.history = [entry, ...this.history].slice(0, HISTORY_CAP);
   471→  }
   472→
   473→  /**
   474→   * Append an activity history entry from an EXTERNAL store (NavStore,
   475→   * MaterialsStore, ToolsStore). This is the SHARED history channel —
   476→   * every admin sub-system logs through here so a single
   477→   * `ActivityHistory` panel on the right side of the page shows entries
   478→   * from every manager.
   479→   *
   480→   * The caller supplies everything except `id` and `at` (we generate
   481→   * those). The entry is prepended, capped, persisted, and emitted in
   482→   * one shot — callers do NOT need to call `persist()`/`emit()`.
   483→   */
   484→  appendHistoryEntry(
   485→    entry: Omit<ActivityHistoryEntry, "id" | "at">
   486→  ) {
   487→    const fullEntry: ActivityHistoryEntry = {
   488→      ...entry,
   489→      id:
   490→        typeof crypto !== "undefined" && "randomUUID" in crypto
   491→          ? crypto.randomUUID()
   492→          : `log-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   493→      at: new Date().toISOString(),
   494→    };
   495→    this.history = [fullEntry, ...this.history].slice(0, HISTORY_CAP);
   496→    this.persist();
   497→    this.emit();
   498→  }

   500→  // ---------- Mutations ----------
   501→  /**
   502→   * Get only ACTIVE banners (not archived). Used for counting against
   503→   * MAX_ACTIVE_BANNERS and for the student app's carousel.
   504→   */
   505→  getActiveBanners(): SponsoredBanner[] {
   506→    return this.banners.filter((b) => b.status !== "archived" && b.enabled !== false);
   507→  }
   508→
   509→  /**
   510→   * Get only ARCHIVED banners.
   511→   */
   512→  getArchivedBanners(): SponsoredBanner[] {
   513→    return this.banners.filter((b) => b.status === "archived");
   514→  }
   515→
   516→  /**
   517→   * Add a new banner. Logs an "uploaded" entry with the banner's thumbnail.
   518→   * Returns the new banner, or null if MAX_ACTIVE_BANNERS has been reached.
   519→   * New banners default to status="active".
   520→   */
   521→  addBanner(input: BannerInput): SponsoredBanner | null {
   522→    if (this.getActiveBanners().length >= MAX_ACTIVE_BANNERS) return null;
   523→    const nextOrder =
   524→      this.banners.length === 0
   525→        ? 1
   526→        : Math.max(...this.banners.map((b) => b.displayOrder)) + 1;
   527→    const banner = makeBanner({ ...input, displayOrder: nextOrder });
   528→    this.banners = [...this.banners, banner];
   529→    this.banners.sort((a, b) => a.displayOrder - b.displayOrder);
   530→    this.log(
   531→      "uploaded",
   532→      banner.title || "بدون عنوان",
   533→      this.bannerThumbnail(banner)
   534→    );
   535→    this.persist();
   536→    this.emit();
   537→    return banner;
   538→  }
   539→
   540→  /**
   541→   * Apply a patch to a banner WITHOUT logging anything.
   542→   *
   543→   * Use this for live preview — when the user is dragging the image
   544→   * positioner or typing in a text field, we want the banner to update
   545→   * in real-time without spamming the activity log with one entry per
   546→   * keystroke. The eventual "edited" log entry is created by
   547→   * `commitBannerEdit` when the user finishes editing.
   548→   */
   549→  updateBannerSilent(id: string, patch: Partial<BannerInput>) {
   550→    let changed = false;
   551→    this.banners = this.banners.map((b) => {
   552→      if (b.id !== id) return b;
   553→      changed = true;
   554→      const updated: SponsoredBanner = {
   555→        ...b,
   556→        ...patch,
   557→        transform: { ...b.transform, ...(patch.transform || {}) },
   558→        updatedAt: new Date().toISOString(),
   559→      };
   560→      return updated;
   561→    });
   562→    if (changed) {
   563→      this.persist();
   564→      this.emit();
   565→    }
   566→  }
   567→
   568→  /**
   569→   * Commit a banner edit: apply the patch AND log exactly ONE "edited"
   570→   * entry with a human-readable change summary.
   571→   *
   572→   * Use this when the user finishes editing (e.g. on blur, on dialog
   573→   * close, on "Save" button click). Pair with `updateBannerSilent` for
   574→   * the live-preview phase.
   575→   *
   576→   * `changeSummary` is a list of Arabic strings describing what changed,
   577→   * e.g. ["العنوان: من 'أ' إلى 'ب'", "تم تعديل الصورة"]. These are
   578→   * stored on the history entry and shown in the UI.
   579→   */
   580→  commitBannerEdit(
   581→    id: string,
   582→    patch: Partial<BannerInput>,
   583→    changeSummary: string[]
   584→  ) {
   585→    let target: SponsoredBanner | undefined;
   586→    this.banners = this.banners.map((b) => {
   587→      if (b.id !== id) return b;
   588→      const updated: SponsoredBanner = {
   589→        ...b,
   590→        ...patch,
   591→        transform: { ...b.transform, ...(patch.transform || {}) },
   592→        updatedAt: new Date().toISOString(),
   593→      };
   594→      target = updated;
   595→      return updated;
   596→    });
   597→    if (target) {
   598→      this.log(
   599→        "edited",
   600→        target.title || "بدون عنوان",
   601→        this.bannerThumbnail(target),
   602→        changeSummary
   603→      );
   604→    }
   605→    this.persist();
   606→    this.emit();
   607→  }
   608→
   609→  /**
   610→   * Delete a banner by id. Logs a "deleted" entry with the banner's
   611→   * thumbnail. Re-sequences displayOrder to be contiguous 1..N.
   612→   */
   613→  deleteBanner(id: string) {
   614→    const target = this.banners.find((b) => b.id === id);
   615→    if (!target) return;
   616→    this.banners = this.banners.filter((b) => b.id !== id);
   617→    this.banners = this.banners
   618→      .sort((a, b) => a.displayOrder - b.displayOrder)
   619→      .map((b, i) => ({ ...b, displayOrder: i + 1 }));
   620→    this.log("deleted", target.title || "بدون عنوان", this.bannerThumbnail(target));
   621→    this.persist();
   622→    this.emit();
   623→  }
   624→
   625→  /**
   626→   * Duplicate a banner by id. The copy gets "(نسخة)" appended to the
   627→   * title and is added at the end. Logs a "duplicated" entry with the
   628→   * source banner's thumbnail.
   629→   * Returns the new banner, or null if MAX_BANNERS has been reached or
   630→   * the source wasn't found.
   631→   */
   632→  duplicateBanner(id: string): SponsoredBanner | null {
   633→    if (this.getActiveBanners().length >= MAX_ACTIVE_BANNERS) return null;
   634→    const source = this.banners.find((b) => b.id === id);
   635→    if (!source) return null;
   636→    const nextOrder =
   637→      this.banners.length === 0
   638→        ? 1
   639→        : Math.max(...this.banners.map((b) => b.displayOrder)) + 1;
   640→    const copy = makeBanner({
   641→      ...source,
   642→      title: `${source.title} (نسخة)`,
   643→      displayOrder: nextOrder,
   644→      transform: { ...source.transform },
   645→      // Duplicates start as active
   646→      status: "active",
   647→      enabled: true,
   648→    });
   649→    this.banners = [...this.banners, copy];
   650→    this.banners.sort((a, b) => a.displayOrder - b.displayOrder);
   651→    this.log("duplicated", source.title || "بدون عنوان", this.bannerThumbnail(source));
   652→    this.persist();
   653→    this.emit();
   654→    return copy;
   655→  }
   656→
   657→  /**
   658→   * Archive a banner — removes it from the carousel but keeps it in
   659→   * the library. Archived banners don't count against MAX_ACTIVE_BANNERS.
   660→   */
   661→  archiveBanner(id: string) {
   662→    const target = this.banners.find((b) => b.id === id);
   663→    if (!target) return;
   664→    this.banners = this.banners.map((b) =>
   665→      b.id === id
   666→        ? { ...b, status: "archived" as const, enabled: false, updatedAt: new Date().toISOString() }
   667→        : b
   668→    );
   669→    this.log("edited", target.title || "بدون عنوان", this.bannerThumbnail(target), [
   670→      "تم أرشفة البانر",
   671→    ]);
   672→    this.persist();
   673→    this.emit();
   674→  }
   675→
   676→  /**
   677→   * Restore an archived banner to active status. Returns false if
   678→   * MAX_ACTIVE_BANNERS has been reached.
   679→   */
   680→  unarchiveBanner(id: string): boolean {
   681→    if (this.getActiveBanners().length >= MAX_ACTIVE_BANNERS) return false;
   682→    const target = this.banners.find((b) => b.id === id);
   683→    if (!target) return false;
   684→    this.banners = this.banners.map((b) =>
   685→      b.id === id
   686→        ? { ...b, status: "active" as const, enabled: true, updatedAt: new Date().toISOString() }
   687→        : b
   688→    );
   689→    this.log("edited", target.title || "بدون عنوان", this.bannerThumbnail(target), [
   690→      "تمت استعادة البانر من الأرشيف",
   691→    ]);
   692→    this.persist();
   693→    this.emit();
   694→    return true;
   695→  }
   696→
   697→  /**
   698→   * Swap a banner's displayOrder with its neighbor in the given
   699→   * direction ("up" = earlier, "down" = later). Logs a "reordered"
   700→   * entry with the moved banner's thumbnail.
   701→   */
   702→  moveBanner(id: string, direction: "up" | "down") {
   703→    const sorted = [...this.banners].sort(
   704→      (a, b) => a.displayOrder - b.displayOrder
   705→    );
   706→    const idx = sorted.findIndex((b) => b.id === id);
   707→    if (idx === -1) return;
   708→    const swapIdx = direction === "up" ? idx - 1 : idx + 1;
   709→    if (swapIdx < 0 || swapIdx >= sorted.length) return;
   710→    const a = sorted[idx];
   711→    const b = sorted[swapIdx];
   712→    // Swap displayOrder
   713→    this.banners = this.banners.map((banner) => {
   714→      if (banner.id === a.id) return { ...banner, displayOrder: b.displayOrder };
   715→      if (banner.id === b.id) return { ...banner, displayOrder: a.displayOrder };
   716→      return banner;
   717→    });
   718→    this.banners.sort((x, y) => x.displayOrder - y.displayOrder);
   719→    this.log("reordered", a.title || "بدون عنوان", this.bannerThumbnail(a));
   720→    this.persist();
   721→    this.emit();
   722→  }
   723→
   724→  /**
   725→   * Empty the activity history array. Persists + emits.
   726→   * Banners are NOT affected.
   727→   */
   728→  clearHistory() {
   729→    this.history = [];
   730→    this.persist();
   731→    this.emit();
   732→  }

   734→  // ---------- Theme ----------
   735→  /**
   736→   * Set the admin dashboard theme ("light" or "dark").
   737→   * No-op if the value is unchanged. Persists to THEME_KEY.
   738→   */
   739→  setAdminTheme(theme: "light" | "dark") {
   740→    if (this.adminTheme === theme) return;
   741→    this.adminTheme = theme;
   742→    this.persist();
   743→    this.emit();
   744→  }
   745→
   746→  // ---------- Carousel settings ----------
   747→  /**
   748→   * Set the carousel auto-slide interval (ms).
   749→   *
   750→   * Only logs an activity entry if the value ACTUALLY changed — this
   751→   * prevents spamming the history when the user drags a slider and
   752→   * releases at the same value.
   753→   *
   754→   * The log entry uses action="settings", bannerTitle="إعدادات الكاروسيل",
   755→   * thumbnail="" (no banner involved), and a change summary describing
   756→   * the old → new duration in seconds.
   757→   */
   758→  setAutoSlideInterval(ms: number) {
   759→    if (!Number.isFinite(ms) || ms <= 0) return;
   760→    if (this.autoSlideInterval === ms) return;
   761→    const oldSec = Math.round(this.autoSlideInterval / 1000);
   762→    const newSec = Math.round(ms / 1000);
   763→    this.autoSlideInterval = ms;
   764→    this.log(
   765→      "settings",
   766→      "إعدادات الكاروسيل",
   767→      "",
   768→      [`مدة التقليب: من ${oldSec} ثانية إلى ${newSec} ثانية`]
   769→    );
   770→    this.persist();
   771→    this.emit();
   772→  }
   773→}
